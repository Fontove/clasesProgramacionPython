# PSEUDOCÓDIGO
# 1. Recibir una lista y un elemento.
# 2. Verificar si el elemento ya existe.
# 3. Si no existe, agregarlo.
# 4. Regresar la lista.

def agregar_unico(lista, elemento):

    if elemento not in lista:
        lista.append(elemento)

    return lista


# Prueba

agregar_unico(numeros, 4)
agregar_unico(numeros, 2)

print(numeros)


# PSEUDOCÓDIGO
# 1. Recibir un diccionario.
# 2. Recibir una llave.
# 3. Recibir un valor.
# 4. Asignar el valor a la llave.
# 5. Si la llave no existe se crea.
# 6. Si existe se reemplaza.

def agregar_valor(diccionario, llave, valor):

    diccionario[llave] = valor

    return diccionario

# PSEUDOCÓDIGO
# 1. Recibir una secuencia de ADN.
# 2. Convertirla a mayúsculas.
# 3. Leer la secuencia de tres en tres.
# 4. Buscar cada codón en el diccionario genético.
# 5. Agregar el aminoácido correspondiente.
# 6. Si el codón no existe, agregar "?".
# 7. Ignorar bases sobrantes si la longitud no es múltiplo de 3.
# 8. Regresar la proteína resultante.

def traducir_proteina(secuencia):

    codigo_genetico = {

        # Fenilalanina
        "TTT":"F", "TTC":"F",

        # Leucina
        "TTA":"L", "TTG":"L",
        "CTT":"L", "CTC":"L",
        "CTA":"L", "CTG":"L",

        # Isoleucina
        "ATT":"I", "ATC":"I",
        "ATA":"I",

        # Metionina (Inicio)
        "ATG":"M",

        # Valina
        "GTT":"V", "GTC":"V",
        "GTA":"V", "GTG":"V",

        # Serina
        "TCT":"S", "TCC":"S",
        "TCA":"S", "TCG":"S",
        "AGT":"S", "AGC":"S",

        # Prolina
        "CCT":"P", "CCC":"P",
        "CCA":"P", "CCG":"P",

        # Treonina
        "ACT":"T", "ACC":"T",
        "ACA":"T", "ACG":"T",

        # Alanina
        "GCT":"A", "GCC":"A",
        "GCA":"A", "GCG":"A",

        # Tirosina
        "TAT":"Y", "TAC":"Y",

        # Histidina
        "CAT":"H", "CAC":"H",

        # Glutamina
        "CAA":"Q", "CAG":"Q",

        # Asparagina
        "AAT":"N", "AAC":"N",

        # Lisina
        "AAA":"K", "AAG":"K",

        # Ácido aspártico
        "GAT":"D", "GAC":"D",

        # Ácido glutámico
        "GAA":"E", "GAG":"E",

        # Cisteína
        "TGT":"C", "TGC":"C",

        # Triptófano
        "TGG":"W",

        # Arginina
        "CGT":"R", "CGC":"R",
        "CGA":"R", "CGG":"R",
        "AGA":"R", "AGG":"R",

        # Glicina
        "GGT":"G", "GGC":"G",
        "GGA":"G", "GGG":"G",

        # Codones de paro
        "TAA":"*",
        "TAG":"*",
        "TGA":"*"
    }

    secuencia = secuencia.upper()

    proteina = ""

    for i in range(0, len(secuencia) - 2, 3):

        codon = secuencia[i:i+3]

        if codon in codigo_genetico:
            proteina += codigo_genetico[codon]
        else:
            proteina += "?"

    return proteina


# Prueba

adn = "ATGGTTCCCTAA"

proteina = traducir_proteina(adn)

print("ADN:", adn)
print("Proteína:", proteina)
