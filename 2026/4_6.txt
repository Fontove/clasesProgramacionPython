def agregar_si_no_existe(lista, elemento):
    """Agrega el elemento solo si no está en la lista"""
    if elemento not in lista:
        lista.append(elemento)
        return True
    return False

def agregar_valor(lista, clave, valor):
    """Si la clave no existe, la agrega. Si existe, sobreescribe su valor"""
    # Buscamos si la clave ya está
    for i in range(0, len(lista), 2):
        if lista[i] == clave:
            lista[i + 1] = valor

    lista.append(clave)
    lista.append(valor)
