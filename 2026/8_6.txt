def agregar_conjunto(lista, elemento):
    if elemento not in lista:
        lista.append(elemento)

def agregar_valor(llaves, valores, llave, valor):
    if llave in llaves:
        posicion = llaves.index(llave)
        valores[posicion] = valor      
    else:
        llaves.append(llave)         
        valores.append(valor)

def traducir(secuencia):
    secuencia = secuencia.upper()
    largo = len(secuencia) - len(secuencia) % 3   
    proteina = ""
    for i in range(0, largo, 3):
        codon = secuencia[i:i+3]
        proteina += codigo[codon]
    return proteina

print(traducir("ATGGCCTAA"))
