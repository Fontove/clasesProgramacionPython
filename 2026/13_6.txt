def agregar_elemento(lista, elemento):
    if elemento not in lista:
        lista.append(elemento)

    return lista

def agregar_valor(llaves, valores, llave, valor):
    if llave in llaves:
        posicion = llaves.index(llave)
        valores[posicion] = valor
    else:
        llaves.append(llave)
        valores.append(valor)

    return llaves, valores

numeros = [1, 2, 3, 4]

agregar_elemento(numeros, 3)
agregar_elemento(numeros, 5)

print("Lista sin elementos repetidos:", numeros)

def traducir_proteina(secuencia):

    secuencia = secuencia.upper()
    proteina = ""

    for i in range(0, len(secuencia) - 2, 3):

        codon = secuencia[i:i+3]

        aminoacido = codigo_genetico.get(codon, "X")

        proteina += aminoacido

    sobrantes = len(secuencia) % 3

    if sobrantes != 0:
        print("Advertencia:", sobrantes,
              "nucleotido(s) sobrante(s) al final.")

    return proteina


adn = "ATGGCCATTGTAATGGGCCGCTGAAAGGGTGCCCGATAG"

resultado = traducir_proteina(adn)

print("Secuencia de ADN:", adn)
print("Proteina:", resultado)
