from pathlib import Path

archivo_fasta = "gene.fna"

organismo = Path(archivo_fasta).stem

total_gc = 0
total_nucleotidos = 0

primera_secuencia = False

def calcular_gc(secuencia):
    cantidad_gc = secuencia.count("G") + secuencia.count("C")
    return cantidad_gc / len(secuencia)

def complementaria(secuencia):
    tabla = str.maketrans("ATGCN", "TACGN")
    return secuencia.translate(tabla)


def formato_gc(valor):
    resultado = f"{int(valor * 100) / 100:.2f}"
    resultado = resultado.rstrip("0").rstrip(".")

    if resultado.startswith("0."):
        resultado = resultado[1:]

    return resultado


with open(archivo_fasta, "r", encoding="utf-8") as archivo:

    for numero_linea, linea in enumerate(archivo, start=1):

        linea = linea.strip()

        if not linea:
            continue

        if linea.startswith(">"):

            encabezado = linea[1:].strip()

            print(f"Organismo {organismo}, contig {encabezado}")

            primera_secuencia = True

        else:

            secuencia = linea.replace(" ", "").upper()

            if not secuencia:
                continue

            gc = calcular_gc(secuencia)

            print(f"GC línea {numero_linea}: {formato_gc(gc)}")

            total_gc += secuencia.count("G") + secuencia.count("C")
            total_nucleotidos += len(secuencia)

            if primera_secuencia:
                print(complementaria(secuencia))
                primera_secuencia = False

            if "ATG" in secuencia:
                print(f"ATG en línea {numero_linea}")


if total_nucleotidos > 0:
    gc_total = total_gc / total_nucleotidos
    print(f"GC total del archivo: {formato_gc(gc_total)}")
else:
    print("El archivo no contiene secuencias.")
