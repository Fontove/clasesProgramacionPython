def isdigit(s):
  if not s:
    return False
  return s.isdigit()


def swapcase(s):
  return s.swapcase()


def capitalize(s):
  return s.capitalize()


def es_par(n):
  return n % 2 == 0
  
assert isdigit("1234") == True
assert isdigit("abcd") == False
assert isdigit("123a") == False
assert isdigit("12a4") == False

assert swapcase("Hola Mundo") == "hOLA mUNDO"
assert capitalize("hola mundo") == "Hola mundo"
assert es_par(2) == True
assert es_par(7) == False

2. count
Input: dos strings, el segundo es sólo un caracter
Output: un entero con el número de veces que aparece el segundo string en el primer string
assert( count("hola mundo","o") == 2 )
assert( count("hola mundo","h") == 1 )
assert( count("hola mundo"," ") == 1 )
assert( count("hola mundo","r") == 0 )

def count(s, sub):
  if sub == "":
    return 0
  return s.count(sub)

assert count("hola mundo", "o") == 2
assert count("hola mundo", "h") == 1
assert count("hola mundo", " ") == 1
assert count("hola mundo", "r") == 0
assert count("hola mundo", "!") == 0
assert count("hola mundo", "") == 0
assert count("", "A") == 0

3. replace
Input: tres strings, el segundo y tercero sólo son un caracter
Output: un string, que sea igual al primero pero reemplazando las ocurrencias del segundo por el tercero
assert( replace("hola","h","H") == "Hola" )
assert( replace("hola","o","O") == "hOla" )
assert( replace("hola","","H") == "hola" )
assert( replace("hola","h","") == "ola" )
assert( replace("holah","h","") == "ola" )


def replace(s, old, new):
  if old == "":
    return s
  return s.replace(old, new)

assert replace("hola", "h", "H") == "Hola"
assert replace("hola", "o", "O") == "hOla"
assert replace("hola", "", "H") == "hola"
assert replace("hola", "h", "") == "ola"
assert replace("holah", "h", "") == "ola"

5. complemento
Input: una string de una secuencia genómica
Output: un string con la secuencia complementaria
assert( complemento("A") == "T" )
assert( complemento("T") == "A" )
assert( complemento("G") == "C" )
assert( complemento("C") == "G" )
assert( complemento("AAAA") == "TTTT" )
assert( complemento("ATGC") == "TACG" )
assert( complemento("") == "" )

def complemento(seq):
  trans = str.maketrans("ATGCatgc", "TACGtacg")
  return seq.translate(trans)

assert complemento("A") == "T"
assert complemento("T") == "A"
assert complemento("G") == "C"
assert complemento("C") == "G"
assert complemento("AAAA") == "TTTT"
assert complemento("ATGC") == "TACG"
assert complemento("") == ""

6. GCcontent
Input: un string con una secuencia de ADN
Output: porcentaje de caracteres que son "G" ó "C"
assert( GCcontent("ATGC") == .5 )
assert( GCcontent("") == 0 )
assert( GCcontent("AAAA") == 0 )
assert( GCcontent("ATgc") == .5 )
assert( GCcontent("GGCC") == 1 )

def GCcontent(seq):
  if not seq:
    return 0.0
  seq_upper = seq.upper()
  gc_count = seq_upper.count("G") + seq_upper.count("C")
  return gc_count / len(seq)

assert GCcontent("ATGC") == 0.5
assert GCcontent("") == 0
assert GCcontent("AAAA") == 0
assert GCcontent("ATgc") == 0.5
assert GCcontent("GGCC") == 1
