def isdigit(texto):
    if texto == "":
        return False

    for caracter in texto:
        if caracter < '0' or caracter > '9':
            return False

    return True


assert isdigit("1234")
assert not isdigit("abcd")
assert not isdigit("123a")
assert not isdigit("12a4")


def swapcase(texto):
    resultado = ""

    for letra in texto:
        if 'a' <= letra <= 'z':
            resultado += chr(ord(letra) - 32)
        elif 'A' <= letra <= 'Z':
            resultado += chr(ord(letra) + 32)
        else:
            resultado += letra

    return resultado


assert swapcase("Hola") == "hOLA"
assert swapcase("PyThOn") == "pYtHoN"


def capitalize(texto):
    if texto == "":
        return ""

    resultado = ""
    primera = texto[0]

    if 'a' <= primera <= 'z':
        resultado += chr(ord(primera) - 32)
    else:
        resultado += primera

    for letra in texto[1:]:
        if 'A' <= letra <= 'Z':
            resultado += chr(ord(letra) + 32)
        else:
            resultado += letra

    return resultado


assert capitalize("hola") == "Hola"
assert capitalize("PYTHON") == "Python"

def replace(texto, viejo, nuevo):
    if viejo == "":
        return texto

    resultado = ""

    for letra in texto:
        if letra == viejo:
            resultado += nuevo
        else:
            resultado += letra

    return resultado


assert replace("hola", "h", "H") == "Hola"
assert replace("hola", "o", "O") == "hOla"
assert replace("hola", "", "H") == "hola"
assert replace("hola", "h", "") == "ola"
assert replace("holah", "h", "") == "ola"

def complemento(secuencia):
    resultado = ""

    for letra in secuencia:
        if letra == "A":
            resultado += "T"
        elif letra == "T":
            resultado += "A"
        elif letra == "G":
            resultado += "C"
        elif letra == "C":
            resultado += "G"

    return resultado


assert complemento("A") == "T"
assert complemento("T") == "A"
assert complemento("G") == "C"
assert complemento("C") == "G"
assert complemento("AAAA") == "TTTT"
assert complemento("ATGC") == "TACG"
assert complemento("") == ""

def GCcontent(secuencia):
    if secuencia == "":
        return 0

    contador = 0

    for letra in secuencia:
        if letra == "G" or letra == "C" or letra == "g" or letra == "c":
            contador += 1

    return contador / len(secuencia)


assert GCcontent("ATGC") == .5
assert GCcontent("") == 0
assert GCcontent("AAAA") == 0
assert GCcontent("ATgc") == .5
assert GCcontent("GGCC") == 1
