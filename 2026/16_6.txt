def traducir_a_proteina(secuencia, diccionario):
    inicio_arn = secuencia.find("AUG")
    if inicio_arn == -1:
        return "No se encontró el codón de inicio (AUG)."
    aminoacidos = []
    for i in range(inicio_arn, len(secuencia), 3):
        codon = secuencia[i:i+3]
        # Para ignorar codón
        if len(codon) < 3:
            break
        aminoacido = diccionario.get(codon, "?")
        aminoacidos.append(aminoacido)
        # Si es codón de stop, se agrega el "*" y se termina
        if aminoacido == "*":
            break
    return aminoacidos
