#Utilice el codigo del E. colli

import os

nombre_fasta = "gene.fna"
organismo = os.path.splitext(nombre_fasta)[0]


def contenido_gc(secuencia):
    if len(secuencia) == 0:
        return 0
    gc = secuencia.count("G") + secuencia.count("C")
    return gc / len(secuencia)


def complemento(secuencia):
    tabla = str.maketrans("ACGT", "TGCA")
    return secuencia.translate(tabla)


archivo = open(nombre_fasta, "r")

total_bases = 0
total_gc = 0
primera_de_segmento = True
numero_linea = 0

for linea in archivo:
    numero_linea += 1
    linea = linea.strip()

    # Saltar líneas vacías
    if linea == "":
        continue

    # Encabezados
    if linea.startswith(">"):
        print(f"Organismo {organismo}, contig {linea[1:]}")
        primera_de_segmento = True
        continue

    secuencia = linea.upper()

    print(f"GC línea {numero_linea}: {contenido_gc(secuencia):.2f}")
    total_bases += len(secuencia)
    total_gc += secuencia.count("G") + secuencia.count("C")

    if primera_de_segmento:
        print(f"Secuencia complementaria: {complemento(secuencia)}")
        primera_de_segmento = False

    if "ATG" in secuencia:
        print(f"ATG en línea {numero_linea}")

archivo.close()

if total_bases > 0:
    print(f"GC total del archivo: {total_gc / total_bases:.2f}")
