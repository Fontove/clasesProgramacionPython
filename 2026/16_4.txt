def complemento(secuencia):
    resultado = ""

    for letra in secuencia:
        if letra == "A":
            resultado += "T"
        elif letra == "T":
            resultado += "A"
        elif letra == "G":
            resultado += "C"
        elif letra == "C":
            resultado += "G"

    return resultado
