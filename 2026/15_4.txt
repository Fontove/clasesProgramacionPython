1. MiEsDigitoy
Input: un string
Output: Booleano indicando si el string se compone de puros números
MiEsDigito("1234") == True # probamos si funciona con puros números
MiEsDigito("abcd") == False # probamos si funciona con puras letras
MiEsDigito("123a") == False # probamos si hay números y letras
MiEsDigito("12a4") == False # probamos que la letra esté a la mitad de los números

Código
def MiEsDigito(digi):
    for caracter in digi:
        if caracter < "0" or caracter > "9":
            return False
    return True

print(MiEsDigito("1234"))
print(MiEsDigito("abcd"))
print(MiEsDigito("123a"))
print(MiEsDigito("12a4"))

Resultados
True
False
False

2. MiConteo
Input: dos strings, el segundo es sólo un caracter
Output: un entero con el número de veces que aparece el segundo string en el primer string
MiConteo("hola mundo","o") == 2
MiConteo("hola mundo","h") == 1
MiConteo("hola mundo"," ") == 1
MiConteo("hola mundo","r") == 0
MiConteo("hola mundo","!") == 0
MiConteo("hola mundo","") == 0
MiConteo("","A") == 0

Código
def MiConteo(texto, caracter):
    contador = 0
    for letra in texto:
        if letra == caracter:
            contador = contador + 1
    return contador

print(MiConteo("hola mundo", "o"))
print(MiConteo("hola mundo", "h"))
print(MiConteo("hola mundo", " "))
print(MiConteo("hola mundo", "r"))
print(MiConteo("hola mundo", "!"))
print(MiConteo("hola mundo", ""))
print(MiConteo("", "A"))

Resultados
2
1
1
0
0
0
0

3. GCcontent
Input: un string con una secuencia de ADN
Output: porcentaje de caracteres que son "G" ó "C"
GCcontent("ATGC") == .5
GCcontent("") == 0
GCcontent("AAAA") == 0
GCcontent("ATgc") == .5
GCcontent("GGCC") == 1

Código
def GCcontent(secuencia):
    if len(secuencia) == 0:
        return 0
    secuencia = secuencia.upper()
    contador = 0
    for base in secuencia:
        if base == "G" or base == "C":
            contador = contador + 1
    return contador / len(secuencia)

print(GCcontent("ATGC"))
print(GCcontent(""))
print(GCcontent("AAAA"))
print(GCcontent("ATgc"))
print(GCcontent("GGCC"))

Resultados
0.5
0
0.0
0.5
1.0

4. MiSwapCase
Input: un string
Output: un string que invierta las mayúsculas y minúsculas

Código
def MiSwapCase(texto):
    resultado = ""
    for letra in texto:
        if letra.isupper():                       
            resultado = resultado + letra.lower()
        elif letra.islower():                     
            resultado = resultado + letra.upper()
        else:                                     
            resultado = resultado + letra
    return resultado

print(MiSwapCase("Hola Mundo")) 
print(MiSwapCase("ABC"))         
print(MiSwapCase("abc"))         
print(MiSwapCase("Hola 123!"))   
print(MiSwapCase(""))  

Resultados
hOLA mUNDO
abc
ABC
hOLA 123!

5. MiCapitalize
Input: un string
Output: un string que tenga la primera letra en mayúsculas

Código
def MiCapitalize(texto):
    if len(texto) == 0:                 
        return texto
    return texto[0].upper() + texto[1:]


print(MiCapitalize("hola mundo"))  
print(MiCapitalize("1hola"))       
print(MiCapitalize(""))            

Resultados
Hola mundo
1hola
