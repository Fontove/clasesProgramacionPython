def MiConteo(b,c):
    count = 0
    for char in b:
        if char == c:
            count+= 1
    return count

#3. GCcontent
def GCcontent (d):
    conteo_gc= 0
    longitud_total = len(d)
    if longitud_total==0:
        return 0.0
    for letra in d:
        if letra== "G" or letra =="C":
            conteo_gc += 1
    porcentaje= (conteo_gc/longitud_total)
    return porcentaje
print(GCcontent("ATGC"))  # Debe imprimir 0.5
print(GCcontent(""))      # Debe imprimir 0.0
print(GCcontent("AAAA"))  # Debe imprimir 0.0
print(GCcontent("ATgc"))  # Debe imprimir 0.5
### Tu función da 0.0 para "ATgc" en lugar de 0.5; ¿qué pasa con las letras minúsculas?
