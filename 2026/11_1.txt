# b. Remover espacios al principio y al final: "  hola" -> "hola"
palabra = "  hola"
palabra = palabra.strip()
print(palabra)  # hola

# c. Invertir mayúsculas y minúsculas: "HoLa" -> "hOlA"
palabra = "HoLa"
palabra = palabra.swapcase()
print(palabra)  # hOlA

# d. Poner la primera letra en mayúscula: "hola" -> "Hola"
palabra = "hola"
palabra = palabra.capitalize()
print(palabra)  # Hola

# 4. ¿Qué hacen estos métodos? Ejemplos:
texto = "banana"
# count(subcadena) cuenta cuántas veces aparece la subcadena.
print(texto.count("an"))  # 2
# find(subcadena) devuelve el índice de la primera aparición.
# Si no la encuentra, devuelve -1 (0 es una posición válida).
print(texto.find("na"))  # 2
print(texto.find("z"))   # -1

# isdigit() devuelve True si todos los caracteres son dígitos y la cadena
# no está vacía; en otro caso devuelve False.
print("123".isdigit())  # True
print("12a".isdigit())  # False

# replace(viejo, nuevo) devuelve una cadena con las apariciones reemplazadas.
print(texto.replace("a", "o"))  # bonono
